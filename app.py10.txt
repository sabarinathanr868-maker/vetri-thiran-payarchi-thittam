    "Ask anything (e.g., 'What if my rent is higher?', 'Can I afford an iPhone?', 'Where should I invest my savings?')",
    placeholder="Type your question here and press enter..."
)

if q_input:
    # Check if user has Gemini API Key or use smart local assistant
    api_key = os.environ.get("GEMINI_API_KEY")
    ai_answer = ""
    
    if api_key:
        try:
            from google import genai
            client = genai.Client(api_key=api_key)
            prompt = f"""
You are PocketSmart AI, an expert, encouraging financial coach.
The user has {CURR}{user_income:,.0f} total money.
Their budget allocation is:
- Needs: {CURR}{amt_needs:,.0f}
- Wants: {CURR}{amt_wants:,.0f}
- Investments: {CURR}{amt_savings:,.0f}
- Emergency: {CURR}{amt_emergency:,.0f}
- Daily allowance: {CURR}{daily_fun_allowance:,.0f}

User Question: {q_input}
Give a concise, direct, helpful, and numerical answer tailored to their exact money numbers.
"""
            res = client.models.generate_content(
                model="gemini-2.5-flash",
                contents=prompt
            )
            ai_answer = res.text
        except Exception:
            pass
            
    if not ai_answer:
        # High quality local contextual answer
        ql = q_input.lower()
        if "rent" in ql or "house" in ql:
            ai_answer = f"If your rent is higher than {CURR}{user_income*0.30:,.0f}, you can adjust by slightly reducing your 'Wants' category (currently {CURR}{amt_wants:,.0f}). Keep your rent + utilities under {CURR}{user_income*0.45:,.0f} to protect your financial stability."
        elif "iphone" in ql or "phone" in ql or "buy" in ql or "gadget" in ql:
            ai_answer = f"For big purchases like a phone or gadget, never take money from your Needs ({CURR}{amt_needs:,.0f}) or Emergency buffer. Instead, save from your monthly Wants allowance ({CURR}{amt_wants:,.0f}/month) over 2 to 4 months, or use the 48-hour cooling rule before buying."
        elif "invest" in ql or "where" in ql or "sip" in ql:
            ai_answer = f"With {CURR}{amt_savings:,.0f}/month to invest, a smart split is: 70% in a broad index fund (Nifty 50 or S&P 500) for long-term compound growth, and 30% in a flexi-cap or balanced fund. Keep your {CURR}{amt_emergency:,.0f} emergency money in high-yield liquid savings."
        else:
            ai_answer = f"Based on your {CURR}{user_income:,.0f} income, the most important action is to automate your {CURR}{amt_savings:,.0f} investment on day one, pay your {CURR}{amt_needs:,.0f} essentials, and stick to your daily fun limit of {CURR}{daily_fun_allowance:,.0f}."
            
    st.info(f"**PocketSmart AI:** {ai_answer}")
